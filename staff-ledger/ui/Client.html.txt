<script>
(function(){
  'use strict';
  var $=function(id){return document.getElementById(id);},R=LedgerRules;
  var view=null,draft=null,pending=null,busy=false,offset=0,query='',storageKey='tarteel.ledger.pending.v1';
  var labels={CREATE_GUARDIAN:'Add guardian',UPDATE_GUARDIAN:'Correct guardian details',CREATE_LEARNER:'Add child',UPDATE_LEARNER:'Correct child details',CREATE_PURCHASE:'Add unpaid package',RECORD_RECEIPT:'Record verified payment',MATCH_PAYMENT:'Match payment to package',SCHEDULE_LESSON:'Schedule lesson or replacement',RECORD_OUTCOME:'Record lesson outcome',RECORD_PAYMENT_ADJUSTMENT:'Record refund or payment reversal',CHANGE_EXPIRY:'Review package expiry',OPEN_PACKAGE:'Open verified historical package',REVERSE_EVENT:'Reverse an incorrect entry',RESOLVE_REVIEW:'Resolve a review case'};
  function node(tag,text,attrs){var e=document.createElement(tag);if(text!==undefined)e.textContent=text;Object.keys(attrs||{}).forEach(function(k){e.setAttribute(k,attrs[k]);});return e;}
  function option(value,text){var e=node('option',text);e.value=value;return e;}
  function today(){return R.madridDate(new Date().toISOString());}
  function rpc(method,arg){return new Promise(function(resolve,reject){var done=false,timer=setTimeout(function(){if(!done){done=true;reject(Error('The response is uncertain. Check save status before continuing.'));}},25000);var run=google.script.run.withSuccessHandler(function(value){if(!done){done=true;clearTimeout(timer);resolve(value);}}).withFailureHandler(function(){if(!done){done=true;clearTimeout(timer);reject(Error('The response is uncertain. Check save status before continuing.'));}});if(arg===undefined)run[method]();else run[method](arg);});}
  function message(text,error,focus){$('status').textContent=text;$('status').classList.toggle('error',Boolean(error));if(focus)$('status').focus();}
  function lock(value){busy=value;$('fields').inert=value;$('task').disabled=value;$('preview-button').disabled=value;$('refresh').disabled=value;$('find').disabled=value;$('previous').disabled=value;$('next').disabled=value;$('confirm').disabled=value;$('edit').disabled=value;$('prepare-summary').disabled=value;$('check-status').hidden=!pending;}
  function error(result){var e=Error(result.error.message);e.code=result.error.code;e.field=result.error.field;throw e;}
  function accepted(result){if(!result.ok)error(result);return result;}
  function fail(e){message(e.message||'The ledger could not complete this change.',true,true);if(e.field&&$('f-'+e.field))$('f-'+e.field).focus();}
  function remember(value){localStorage.setItem(storageKey,JSON.stringify(value));pending=value;}
  function forget(){localStorage.removeItem(storageKey);pending=null;}
  function description(key,row){if(key==='students')return row.firstName+' · '+row.learnerId;if(key==='guardians')return row.name+' · '+row.guardianId;if(key==='packages')return row.child+' · '+row.name+' · '+row.status+' · '+row.purchaseId;if(key==='payments')return row.guardian+' · €'+row.amountEUR.toFixed(2)+' · '+row.paymentId;if(key==='lessons')return row.child+' · '+row.startsAt+' · '+row.status+' · '+row.lessonId;if(key==='entitlements')return row.status+' · '+row.entitlementId;return row.required.join(', ')+' · '+row.caseId;}
  function lists(){['guardians','students','packages','payments','lessons','entitlements','reviewCases'].forEach(function(key){var old=$('choices-'+key);if(old)old.remove();var list=node('datalist',undefined,{id:'choices-'+key}),idKey={guardians:'guardianId',students:'learnerId',packages:'purchaseId',payments:'paymentId',lessons:'lessonId',entitlements:'entitlementId',reviewCases:'caseId'}[key];view[key].forEach(function(row){list.append(option(row[idKey],description(key,row)));});document.body.append(list);});}
  function render(){
    $('identity').textContent='Signed in: '+view.actorEmail+' · Applied event '+view.health.appliedSequence+' / '+view.sequence;
    $('overview').replaceChildren();if(view.health.stale)$('overview').append(node('p','Views need refreshing. Saved events are preserved.'));
    if(!view.overview.exceptions.length)$('overview').append(node('p','No outstanding items.'));
    view.overview.exceptions.forEach(function(row){var card=node('div',undefined,{class:'card'});card.append(node('strong',row.type),node('p',row.details),node('p',row.action),node('code',row.recordId));$('overview').append(card);});
    $('packages').replaceChildren();if(!view.packages.length)$('packages').append(node('p','No packages on this page.'));
    view.packages.forEach(function(p){var card=node('div',undefined,{class:'card'});card.append(node('strong',p.child+' · '+p.name),node('p',p.status+' · €'+p.matchedEUR.toFixed(2)+' matched / €'+p.priceEUR.toFixed(2)),node('p',p.remaining+' remaining · '+p.reserved+' reserved · '+p.available+' available'),node('p',p.makeupsAvailable+' make-ups available · Last usable '+p.lastUsableOn),node('code',p.purchaseId));$('packages').append(card);});
    lists();$('record-list').replaceChildren();['guardians','students','packages','payments','lessons','entitlements','reviewCases'].forEach(function(key){view[key].forEach(function(row){$('record-list').append(node('p',description(key,row)));});});
    $('page-info').textContent='Page '+(offset/100+1)+' · Up to 100 per record type · '+Object.entries(view.page.totals).map(function(p){return p[0]+': '+p[1];}).join(', ');
  }
  async function load(){view=accepted(await rpc('ledgerBootstrap',{offset:offset,limit:100,query:query}));render();}
  function field(key,label,type,opts){opts=opts||{};var wrap=node('div'),id='f-'+key,input;
    if(type==='select'){input=node('select');opts.options.forEach(function(pair){input.append(option(pair[0],pair[1]));});}
    else if(type==='textarea'){input=node('textarea');input.rows=3;}
    else{input=node('input');input.type=type==='money'?'text':type;}
    input.id=id;input.name=key;input.required=!opts.optional;input.maxLength=opts.max||100;if(opts.list)input.setAttribute('list','choices-'+opts.list);
    if(type==='money')input.inputMode='decimal';if(type==='number'){input.min='0';input.step='1';}if(type==='checkbox')input.required=false;
    if(opts.value!==undefined)input.value=opts.value;else if(type==='date')input.value=today();else if(type==='number')input.value='0';
    var lab=node('label',label,{'for':id});if(type==='checkbox'){lab.prepend(input);lab.className='check';wrap.append(lab);}else wrap.append(lab,input);
    if(opts.hint){var hint=node('p',opts.hint,{id:id+'-hint',class:'hint'});input.setAttribute('aria-describedby',hint.id);wrap.append(hint);}
    $('form-fields').append(wrap);return input;
  }
  function selectField(key,label,values){return field(key,label,'select',{options:values.map(function(v){return Array.isArray(v)?v:[v,v.replaceAll('_',' ').toLowerCase()];})});}
  function record(key,label,list,optional){return field(key,label,'text',{list:list,optional:optional,max:36,hint:'Choose a record suggestion or paste its full ID from the workbook. Use Find a record to search other pages.'});}
  function datetime(key,label,optional){return field(key,label,'datetime-local',{optional:optional,hint:'Europe/Madrid local time. For an autumn repeated hour, choose its clock offset below.'});}
  function commitments(){
    var box=node('div',undefined,{id:'commitments'}),button=node('button','Add commitment',{type:'button'});$('form-fields').append(box,button);
    button.addEventListener('click',function(){var n=box.children.length+1,card=node('fieldset',undefined,{class:'commitment'});card.dataset.id=crypto.randomUUID();card.dataset.lessonId=crypto.randomUUID();card.append(node('legend','Outstanding commitment '+n));
      function input(key,label,type,choices){var id='commitment-'+n+'-'+key,control=node(type==='select'?'select':'input',undefined,{id:id,'data-key':key});if(type==='select')choices.forEach(function(v){control.append(option(v[0],v[1]));});else control.type=type;control.maxLength=128;var lab=node('label',label,{'for':id});card.append(lab,control);return control;}
      input('sourceReference','Commitment reference','text');var kind=input('kind','Commitment type','select',[['MAKEUP_DUE','Replacement due; no date yet'],['SCHEDULED','Lesson already scheduled']]);
      var teacher=input('teacher','Commitment teacher','text'),starts=input('startsAt','Commitment start (Madrid)','datetime-local'),replacement=input('isReplacement','Scheduled lesson is a replacement','checkbox');
      function toggle(){[teacher,starts,replacement].forEach(function(c){c.disabled=kind.value!=='SCHEDULED';});}kind.addEventListener('change',toggle);toggle();
      var remove=node('button','Remove commitment',{type:'button'});remove.addEventListener('click',function(){card.remove();});card.append(remove);box.append(card);
    });
  }
  function drawForm(){
    draft=null;$('confirmation').hidden=true;$('form-fields').replaceChildren();var type=$('task').value;$('form-title').textContent=labels[type];
    function name(){field('name','Guardian name','text');field('email','Guardian email','email',{max:254});field('whatsapp','Guardian WhatsApp','tel',{max:40,hint:'International format, for example +34600000000.'});}
    function learner(){record('learnerId','Child','students');}
    function dates(){selectField('packageCode','Package',[['STARTER','Starter — €120 · 6 lessons · 0 make-ups'],['STEADY','Steady — €220 · 12 lessons · 4 make-ups'],['COMMITTED','Committed — €400 · 25 lessons · 4 make-ups']]);var p=field('purchasedOn','Purchase date','date'),last=field('lastUsableOn','Last usable date (Madrid)','date',{value:R.defaultLastUsableOn(today())});p.addEventListener('input',function(){try{last.value=R.defaultLastUsableOn(p.value);}catch{}});}
    if(type==='CREATE_GUARDIAN'||type==='UPDATE_GUARDIAN'){if(type==='UPDATE_GUARDIAN')record('guardianId','Guardian','guardians');name();}
    if(type==='CREATE_LEARNER'||type==='UPDATE_LEARNER'){if(type==='CREATE_LEARNER')record('guardianId','Guardian','guardians');else learner();field('firstName','Child name','text');selectField('language','Teaching language',['english','arabic','turkish','persian']);if(type==='CREATE_LEARNER')field('bookingReference','Booking reference (optional)','text',{optional:true,max:128});}
    if(type==='CREATE_PURCHASE'||type==='OPEN_PACKAGE'){learner();dates();}
    if(type==='RECORD_RECEIPT'){record('guardianId','Guardian','guardians');field('source','Payment source','text',{max:64});field('reference','Verified payment reference','text',{max:128});field('receivedOn','Received date','date');field('amountEUR','Amount (EUR)','money');field('verified','I checked this payment in the bank or payment service','checkbox');}
    if(type==='MATCH_PAYMENT'){record('paymentId','Verified payment','payments');record('purchaseId','Package purchase','packages');field('amountEUR','Amount to match (EUR)','money');}
    if(type==='SCHEDULE_LESSON'){learner();selectField('kind','Lesson type',[['PAID','Paid — choose package'],['TRIAL','Trial — no package']]);record('purchaseId','Package purchase','packages',true);record('entitlementId','Existing reservation (replacement only)','entitlements',true);field('teacher','Teacher','text');datetime('startsAt','Start (Madrid)');}
    if(type==='RECORD_OUTCOME'){record('lessonId','Lesson','lessons');selectField('outcome','Outcome',['COMPLETED','PARENT_CANCELLED','SERVICE_CANCELLED','NO_SHOW','SHORTENED','TECHNICAL_FAILURE','DISPUTED']);datetime('noticeAt','Parent notice (Madrid)',true);}
    if(type==='RECORD_PAYMENT_ADJUSTMENT'){record('originalReceiptId','Original verified payment','payments');selectField('kind','Adjustment type',['REFUND','REVERSAL']);field('amountEUR','Amount (EUR)','money');field('occurredOn','Adjustment date','date');field('reference','Verified adjustment reference','text',{max:128});field('verified','I verified this refund or reversal','checkbox');}
    if(type==='CHANGE_EXPIRY'){record('purchaseId','Package purchase','packages');field('lastUsableOn','New last usable date (Madrid)','date');}
    if(type==='OPEN_PACKAGE'){['completed','forfeited','makeupsUsed'].forEach(function(k){field(k,{completed:'Completed lessons',forfeited:'Charged absences',makeupsUsed:'Make-ups already used'}[k],'number');});field('sourceReference','Opening source reference','text',{max:128});field('paymentReference','Historical payment evidence reference','text',{max:128});field('verifiedOn','Payment evidence checked on','date');field('verified','I verified the historical package payment','checkbox');selectField('commitmentStatus','Outstanding commitments',[['unknown','Unknown — hold this package for review'],['none','Checked — none outstanding'],['listed','Checked — all commitments listed below']]);commitments();}
    if(type==='REVERSE_EVENT')record('originalEventId','Original event ID (Events tab)',null);
    if(type==='RESOLVE_REVIEW'){record('caseId','Review case','reviewCases');selectField('decision','Review decision',['RECONCILIATION_CONFIRMED','MARK_COMPLETED','CHARGE_ABSENCE','PRESERVE_REPLACEMENT','EXPIRY_CONFIRMED','COMMITMENTS_CONFIRMED','CLOSE_PACKAGE']);field('lastUsableOn','Confirmed last usable date (expiry review)','date');commitments();}
    if(['SCHEDULE_LESSON','RECORD_OUTCOME','OPEN_PACKAGE','RESOLVE_REVIEW'].includes(type))selectField('clockOffset','Clock offset',[['auto','Automatic when unambiguous'],['+02:00','Summer time (UTC+02:00)'],['+01:00','Winter time (UTC+01:00)']]);
    field('reason','Reason / staff note','textarea',{optional:!(/^(UPDATE_|CHANGE_|REVERSE_|RESOLVE_|OPEN_|RECORD_PAYMENT_ADJUSTMENT)/.test(type)),max:1000,hint:'Required for corrections, reviews, refunds and historical openings. Keep notes factual.'});
    $('change-form').dataset.targetId=crypto.randomUUID();
  }
  function madridInstant(value){
    var choice=$('f-clockOffset').value,offsets=choice==='auto'?['+01:00','+02:00']:[choice],matches=offsets.map(function(o){return value+':00'+o;}).filter(function(i){if(!Number.isFinite(Date.parse(i)))return false;var p=new Intl.DateTimeFormat('sv-SE',{timeZone:'Europe/Madrid',year:'numeric',month:'2-digit',day:'2-digit',hour:'2-digit',minute:'2-digit',hourCycle:'h23'}).format(new Date(i)).replace(' ','T');return p===value;});
    if(matches.length!==1)throw Error('That Madrid time is missing or repeated at the clock change. Check the time and choose the correct clock offset.');return new Date(matches[0]).toISOString();
  }
  function readCommitments(){return Array.from($('commitments').children).map(function(card){function v(k){return card.querySelector('[data-key="'+k+'"]').value;}var c={entitlementId:card.dataset.id,sourceReference:v('sourceReference'),kind:v('kind')};if(c.kind==='SCHEDULED')Object.assign(c,{lessonId:card.dataset.lessonId,teacher:v('teacher'),startsAt:madridInstant(v('startsAt')),timeZone:'Europe/Madrid',isReplacement:card.querySelector('[data-key="isReplacement"]').checked});return c;});}
  function command(){
    var type=$('task').value,p={},extra=['reason','clockOffset','amountEUR','paymentReference','verifiedOn','commitmentStatus'];
    Array.from($('form-fields').querySelectorAll('[id^="f-"]')).forEach(function(e){var k=e.name;if(extra.includes(k)||(!e.value&&e.type!=='checkbox'))return;p[k]=e.type==='checkbox'?e.checked:e.type==='number'?Number(e.value):e.value;});
    var idKey={CREATE_GUARDIAN:'guardianId',CREATE_LEARNER:'learnerId',CREATE_PURCHASE:'purchaseId',RECORD_RECEIPT:'paymentId',MATCH_PAYMENT:'matchId',SCHEDULE_LESSON:'lessonId',RECORD_PAYMENT_ADJUSTMENT:'adjustmentId',OPEN_PACKAGE:'purchaseId'}[type];if(idKey)p[idKey]=$('change-form').dataset.targetId;
    if($('f-amountEUR'))p.amountCents=R.parseEuroAmount($('f-amountEUR').value);
    if(type==='RECORD_RECEIPT')p.currency='EUR';
    if(type==='SCHEDULE_LESSON'){p.startsAt=madridInstant(p.startsAt);p.timeZone='Europe/Madrid';if(p.kind==='TRIAL'){delete p.purchaseId;delete p.entitlementId;}}
    if(type==='RECORD_OUTCOME'){if(p.outcome==='PARENT_CANCELLED')p.noticeAt=madridInstant(p.noticeAt);else delete p.noticeAt;}
    if(type==='OPEN_PACKAGE'){p.commitments=readCommitments();var mode=$('f-commitmentStatus').value;if(mode==='none'&&p.commitments.length||mode==='listed'&&!p.commitments.length)throw Error('Check the commitment selection and list.');p.commitmentsConfirmed=mode!=='unknown';p.paymentEvidence={verified:p.verified,verifiedOn:$('f-verifiedOn').value,sourceReference:$('f-paymentReference').value};delete p.verified;}
    if(type==='RESOLVE_REVIEW'){p.decisionData=p.decision==='EXPIRY_CONFIRMED'?{lastUsableOn:p.lastUsableOn}:p.decision==='COMMITMENTS_CONFIRMED'?{commitments:readCommitments()}:{};delete p.lastUsableOn;}
    return R.normalizeCommand({schemaVersion:1,requestId:crypto.randomUUID(),expectedSequence:view.sequence,type:type,payload:p,reason:$('f-reason').value});
  }
  function previewText(c,p){
    $('preview').replaceChildren(node('p',labels[c.type]));Object.entries(c.payload).forEach(function(pair){var value=pair[1];if(pair[0]==='amountCents')value='€'+(value/100).toFixed(2);else if(pair[0]==='commitments')value=value.length+' outstanding commitments';else if(pair[0]==='paymentEvidence')value='Verified historical payment · '+value.sourceReference;else if(pair[0]==='decisionData')value=value.lastUsableOn||((value.commitments||[]).length+' commitments');$('preview').append(node('p',pair[0].replace(/([A-Z])/g,' $1')+': '+value));});if(c.reason)$('preview').append(node('p','Reason: '+c.reason));
    p.balanceChanges.forEach(function(change){var b=change.after;$('preview').append(node('p','Result: '+b.remaining+' remaining · '+b.reserved+' reserved · '+b.available+' available · '+b.makeupsAvailable+' make-ups available'+(b.held?' · Under review':'')));});
    p.createdReviewCases.forEach(function(review){$('preview').append(node('p','Needs review: '+review.required.join(', ')));});
  }
  async function preview(e){e.preventDefault();if(busy||pending)return;try{var c=command();lock(true);message('Checking this change…');var result=accepted(await rpc('ledgerPreview',c));draft=c;previewText(c,result.preview);$('confirmation').hidden=false;lock(false);$('fields').inert=true;$('task').disabled=true;$('preview-button').disabled=true;$('confirm-title').focus();message('Preview ready. Check all details before saving.');}catch(e){lock(false);fail(e);}}
  async function save(){if(busy||!draft||pending)return;try{
    lock(true);
    var bytes=new TextEncoder().encode(R.canonicalCommand(draft)),digest=await crypto.subtle.digest('SHA-256',bytes),fingerprint=Array.from(new Uint8Array(digest)).map(function(v){return v.toString(16).padStart(2,'0');}).join('');
    remember({requestId:draft.requestId,targetId:$('change-form').dataset.targetId,fingerprint:fingerprint});lock(true);message('Saving… Keep this window open.');var result=await rpc('ledgerExecute',draft);if(!result.ok&&result.error.code==='SAVE_UNCERTAIN')throw Error(result.error.message);accepted(result);await saved(result);
  }catch(e){if(pending&&(!e.code||e.code==='SAVE_UNCERTAIN')){lock(true);$('check-status').disabled=false;message('Save uncertain. Request '+pending.requestId+'. Check save status before continuing.',true,true);}else{forget();draft=null;$('confirmation').hidden=true;lock(false);fail(e);}}}
  async function saved(result){forget();draft=null;$('confirmation').hidden=true;lock(false);await load();drawForm();message(result.status==='SAVED_REFRESH_REQUIRED'?'Saved — refresh required. Do not enter this change again.':'Saved · Event '+result.sequence+'.',false,true);}
  async function status(){if(!pending)return;try{lock(true);message('Checking save '+pending.requestId+'…');var result=accepted(await rpc('ledgerCommandStatus',pending.requestId));if(result.status==='SAVED'||result.status==='SAVED_REFRESH_REQUIRED'){await saved(result);return;}if(result.status==='NOT_FOUND'){forget();draft=null;lock(false);$('confirmation').hidden=true;await load();message('No saved event found. Review the details and preview a new change.',false,true);}else{lock(true);message('Save uncertain. Request '+pending.requestId+'. Use Tarteel Ledger → Review an uncertain save after inspecting the original execution.',true,true);}}catch(e){lock(true);fail(e);}}
  Object.entries(labels).forEach(function(p){$('task').append(option(p[0],p[1]));});$('task').addEventListener('change',drawForm);$('change-form').addEventListener('submit',preview);$('confirm').addEventListener('click',save);$('check-status').addEventListener('click',status);
  $('edit').addEventListener('click',function(){draft=null;$('confirmation').hidden=true;lock(false);$('form-fields').querySelector('input,select,textarea').focus();});
  $('refresh').addEventListener('click',async function(){if(pending)return;try{lock(true);accepted(await rpc('ledgerRefresh'));await load();draft=null;$('confirmation').hidden=true;message('Refreshed. Preview again before saving.');}catch(e){fail(e);}finally{lock(false);}});
  async function find(){if(busy||pending)return;try{lock(true);await load();message('Records loaded.');}catch(e){fail(e);}finally{lock(false);}}
  $('find').addEventListener('click',function(){offset=0;query=$('search').value;find();});$('previous').addEventListener('click',function(){offset=Math.max(0,offset-100);find();});$('next').addEventListener('click',function(){if(Object.values(view.page.totals).some(function(n){return n>offset+100;})){offset+=100;find();}});
  $('prepare-summary').addEventListener('click',async function(){if(busy||pending)return;try{var id=R.uuid($('summary-child').value);lock(true);var r=accepted(await rpc('ledgerParentSummary',id));$('summary').value=r.summary;message('Summary prepared. Check the child’s name before sharing.');}catch(e){$('summary').value='';fail(e);}finally{lock(false);}});
  $('copy-summary').addEventListener('click',async function(){if(!$('summary').value)return;try{await navigator.clipboard.writeText($('summary').value);message('Summary copied.');}catch{$('summary').focus();$('summary').select();message('Select and copy the prepared summary.');}});
  async function start(){try{if(!window.google||!google.script){await new Promise(function(resolve,reject){var script=node('script',undefined,{src:'/__ledger/bridge.js'});script.onload=resolve;script.onerror=reject;document.head.append(script);});}lock(true);await load();drawForm();var raw=localStorage.getItem(storageKey);if(raw){pending=JSON.parse(raw);R.uuid(pending.requestId);await status();}else{lock(false);message('Ready. Choose a task to record a change.');}}catch(e){lock(true);fail(e);}}
  start();
})();
